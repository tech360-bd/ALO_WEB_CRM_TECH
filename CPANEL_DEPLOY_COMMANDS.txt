# 1. CREATE CPANEL DIRECTORY STRUCTURE (Run via SSH in /home/aloeduca)
cd /home/aloeduca

mkdir -p \
  alo-web-crm \
  alo-web-crm-shared/backups \
  alo-web-crm-shared/uploads \
  alo-web-crm-shared/logs \
  alo-web-crm-shared/private-documents \
  alo_private_delivery

# 2. SET PERMISSIONS
chmod 700 /home/aloeduca/alo-web-crm-shared
chmod 700 /home/aloeduca/alo_private_delivery
chmod 600 /home/aloeduca/alo-web-crm-shared/.env 2>/dev/null || true

# 3. UNPACK PRODUCTION DEPLOYMENT PACKAGE
cd /home/aloeduca/alo-web-crm
unzip -o /home/aloeduca/alo_private_delivery/ALO_EDUCATION_PRODUCTION_DEPLOYMENT.zip

# 4. SYMLINK PRIVATE ENVIRONMENT FILE
ln -sfn /home/aloeduca/alo-web-crm-shared/.env /home/aloeduca/alo-web-crm/.env

# 5. RUN DATABASE MIGRATIONS (MySQL 8.0.37)
npx prisma generate
npx prisma migrate deploy

# 6. FILE PERMISSIONS FOR PRODUCTION RUNTIME
find /home/aloeduca/alo-web-crm -type d -exec chmod 755 {} \;
find /home/aloeduca/alo-web-crm -type f -exec chmod 644 {} \;
chmod 750 /home/aloeduca/alo-web-crm/scripts/*.sh 2>/dev/null || true

# 7. START / RESTART NODE.JS APP IN CPANEL PASSING PORT
touch /home/aloeduca/alo-web-crm/tmp/restart.txt
