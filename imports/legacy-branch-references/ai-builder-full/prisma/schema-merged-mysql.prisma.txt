  courseSlug          String? // set when entityType = COURSE
  countrySlug         String? // set when entityType = COUNTRY / GOVERNMENT / AUTHORITY
  // Source metadata
  sourceUrl           String    @unique // the official URL to monitor
  sourceName          String // human label e.g. "University of Oxford — Course Catalogue"
  sourceType          String    @default("UNIVERSITY_OFFICIAL") // UNIVERSITY_OFFICIAL | GOVERNMENT | AUTHORITY | ADMISSIONS_PORTAL | PDF | APPROVED_INSTITUTIONAL
  category            String? // CATALOGUE | ADMISSIONS | SCHOLARSHIPS | VISAS | FEES | INTAKES | REQUIREMENTS | POLICY
  priority            Int       @default(1) // 1 = official university, 2 = government, 3 = authority, 4 = portal, 5 = PDF, 6 = approved institutional
  // Monitoring config
  monitoringEnabled   Boolean   @default(true)

  lastContentHash     String? // SHA-256 of normalized page content

  status              String    @default("ACTIVE") // ACTIVE | PAUSED | RETIRED | BROKEN

  @@index([courseSlug])

  @@index([priority])

  contentLength     Int            @default(0)
  title             String? // <title> or detected heading
  contentType       String? // text/html, application/pdf, etc.

  // Full raw HTML is optionally stored in object storage in production.
  textExtract       String // normalized plain text (truncated to 50KB)

  @@index([sourceId])
  @@index([capturedAt])
}

// A detected change between two snapshots. One change record per field that changed.
model DataChange {
  id                 String         @id @default(cuid())
  runId              String // VerificationRun id
  sourceId           String
  source             OfficialSource @relation(fields: [sourceId], references: [id], onDelete: Cascade)
  // What entity changed
  entityType         String // UNIVERSITY | COURSE | FACULTY | SCHOLARSHIP | GOVERNMENT_POLICY
  universitySlug     String?
  courseSlug         String?
  countrySlug        String?
  // Change classification (one of the 27 change types from the spec)
  changeType         String // NEW_UNIVERSITY | UNIVERSITY_UPDATED | UNIVERSITY_CLOSED | NEW_FACULTY | FACULTY_UPDATED | NEW_COURSE | COURSE_UPDATED | COURSE_CLOSED | COURSE_RENAMED | COURSE_CODE_CHANGED | FEE_CHANGED | INTAKE_CHANGED | DEADLINE_CHANGED | REQUIREMENT_CHANGED | ENGLISH_REQUIREMENT_CHANGED | SCHOLARSHIP_NEW | SCHOLARSHIP_UPDATED | SCHOLARSHIP_CLOSED | DURATION_CHANGED | CAMPUS_CHANGED | STUDY_MODE_CHANGED | APPLICATION_PROCESS_CHANGED | OFFICIAL_POLICY_CHANGE | GOVERNMENT_POLICY_CHANGE | SOURCE_CHANGED | SOURCE_UNAVAILABLE | POSSIBLE_DATA_CONFLICT | NEEDS_MANUAL_REVIEW
  fieldName          String // e.g. "tuition", "intake", "deadline", "englishRequirement"
  // Values (preserved for history — NEVER destroy old value)
  oldValue           String? // previous verified value (NULL if field didn't exist before)
  newValue           String? // newly detected value (NULL if field removed)
  // Evidence
  sourceUrl          String // official URL where the change was detected
  sourcePageTitle    String?
  detectedAt         DateTime       @default(now())
  previousVerifiedAt DateTime? // when the OLD value was last verified
  newSourceDate      String? // publication date if detectable from the source page
  // AI analysis
  aiExplanation      String? // human-readable AI explanation of the change
  confidence         Float          @default(0) // 0.0–1.0
  impactLevel        String         @default("LOW") // LOW | MEDIUM | HIGH | CRITICAL
  // Approval workflow
  status             String         @default("PENDING") // PENDING | APPROVED | REJECTED | NEEDS_REVIEW | RECHECK_REQUESTED | SUPERSEDED
  reviewedBy         String? // userId of Super Admin
  reviewedAt         DateTime?
  reviewNotes        String?
  // When approved, the production database mutation is applied and recorded here
  appliedAt          DateTime?
  appliedBy          String?
  // Supersession: if a newer change supersedes this one, link it
  supersededBy       String?
  createdAt          DateTime       @default(now())
  updatedAt          DateTime       @updatedAt

  values    DataChangeValue[]
  approvals ApprovalAction[]

  @@index([runId])
  @@index([sourceId])
  @@index([entityType])
  @@index([universitySlug])
  @@index([courseSlug])
  @@index([changeType])
  @@index([status])
  @@index([impactLevel])
  @@index([detectedAt])
}

// Server-side file storage with three pools, checksums, audit
// ============================================================

// Central file metadata record. Every file uploaded, generated,
// or received by the platform gets a FileMetadata row. The actual
// file bytes live on disk in the configured storage pool; the DB
// stores the metadata + checksum + access policy + audit trail.
model FileMetadata {
  id                 String         @id @default(cuid())
  uuid               String         @unique // UUID-based storage name (never trust client filename)
  originalName       String // client-provided filename (sanitized)
  storedName         String // actual on-disk filename (uuid + extension)
  mimeType           String // server-validated MIME type (never trust client)
  extension          String // lowercase extension without dot
  sizeBytes          BigInt // verified file size
  checksumSha256     String // SHA-256 of file content (for integrity + dedup)
  // Storage routing
  storagePool        String // STUDENT_PARTNER | OFFICIAL | OPERATIONAL
  storagePath        String // relative path within pool, e.g. "student/STU-000001/documents/passport/uuid.pdf"
  absolutePath       String? // computed absolute path (for server-side access)
  // Entity linkage (who/what this file belongs to)
  entityType         String // STUDENT | PARTNER | EMPLOYEE | APPLICATION | EXAM | INTERVIEW | CAMPAIGN | OFFICIAL | PROJECT | LEAD | COMMUNICATION | REVIEW | BRAND
  entityId           String? // the entity ID (studentId, partnerId, etc.)
  studentId          String?
  partnerId          String?
  employeeId         String?
  applicationId      String?
  projectId          String?
  // Upload context
  uploadedBy         String? // userId
  uploadedAt         DateTime       @default(now())
  uploadedFromIp     String?
  source             String         @default("WEB_UPLOAD") // WEB_UPLOAD | API | WHATSAPP | EMAIL | AI_GENERATED | IMPORT | EXTERNAL
  // Access control
  visibility         String         @default("STUDENT_VISIBLE") // STUDENT_VISIBLE | INTERNAL_ONLY | PARTNER_VISIBLE | B2B_CONFIDENTIAL | PUBLIC | ARCHIVED
  accessPolicy       String         @default("OWNER_AND_STAFF") // OWNER_ONLY | OWNER_AND_STAFF | STAFF_ONLY | PARTNER_ONLY | PUBLIC | SUPER_ADMIN_ONLY
  // Lifecycle
  status             String         @default("ACTIVE") // ACTIVE | QUARANTINED | ARCHIVED | DELETED
  isDeleted          Boolean        @default(false)
  deletedAt          DateTime?
  deletedBy          String?
  deletionReason     String?
  // Versioning
  version            Int            @default(1)
  previousVersionId  String? // link to previous version if this is a replacement
  previousVersion    FileMetadata?  @relation("FileVersions", fields: [previousVersionId], references: [id])
  versions           FileMetadata[] @relation("FileVersions")
  // Retention
  retentionPolicy    String? // STUDENT_PERMANENT | EMPLOYEE_5Y | MARKETING_2Y | PROJECT_PERMANENT | EXAM_PERMANENT
  retentionExpiresAt DateTime?
  // Metadata
  metadata           String? // JSON: {dimensions, duration, pages, altText, ...}
  tags               String? // JSON array of string tags
  createdAt          DateTime       @default(now())
  updatedAt          DateTime       @updatedAt

  events FileEvent[]

  @@index([storagePool])
  @@index([entityType])
  @@index([entityId])
  @@index([studentId])
  @@index([partnerId])
  @@index([employeeId])
  @@index([applicationId])
  @@index([uploadedBy])
  @@index([status])
  @@index([visibility])
  @@index([checksumSha256])
  @@index([createdAt])
}

// Audit trail for every file event (upload/download/view/delete/etc).
model FileEvent {
  id              String       @id @default(cuid())
  fileId          String
  file            FileMetadata @relation(fields: [fileId], references: [id], onDelete: Cascade)
  action          String // UPLOAD | DOWNLOAD | VIEW | SHARE | MOVE | COPY | VERSION | ARCHIVE | RESTORE | DELETE | FAILED_UPLOAD | QUARANTINE
  performedBy     String? // userId
  performedAt     DateTime     @default(now())
  performedFromIp String?
  result          String       @default("SUCCESS") // SUCCESS | FAILURE | DENIED
  details         String? // JSON: {reason, error, ...}
  createdAt       DateTime     @default(now())

  @@index([fileId])
  @@index([action])
  @@index([performedBy])
  @@index([createdAt])
}

// Storage pool configuration (capacity, usage, warnings).
model StoragePool {
  id                String    @id @default(cuid())
  poolKey           String    @unique // STUDENT_PARTNER | OFFICIAL | OPERATIONAL
  name              String // "Student & Partner Private Storage"
  description       String?
  mountPath         String // absolute path on server, e.g. "/home/alo/storage/student-partner"
  capacityBytes     BigInt    @default(0) // 0 = unlimited
  usedBytes         BigInt    @default(0)
  fileCount         Int       @default(0)
  warningThreshold  Int       @default(70) // percent
  highThreshold     Int       @default(85)
  criticalThreshold Int       @default(95)
  status            String    @default("HEALTHY") // HEALTHY | WARNING | HIGH | CRITICAL | OFFLINE
  lastHealthCheck   DateTime?
  createdAt         DateTime  @default(now())
  updatedAt         DateTime  @updatedAt

  @@index([poolKey])
}
