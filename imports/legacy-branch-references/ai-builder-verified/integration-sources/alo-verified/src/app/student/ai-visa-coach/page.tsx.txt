  Bot, FileText, Search, Sparkles, Mic, FileSignature,

} from 'lucide-react'

} from '@/components/ui/dialog'

type CoachAction =
  | 'summarize_file'
  | 'detect_missing_documents'
  | 'suggest_next_action'
  | 'create_interview_questions'
  | 'draft_sop'
  | 'highlight_financial_risk'
  | 'highlight_refusal_risk'
  | 'generate_counselor_summary'
  | 'generate_visa_officer_checklist'

interface ActionDef {
  key: CoachAction
  title: string
  description: string
  icon: typeof Bot
  requiresOtp: boolean
  color: string
  bg: string
}

interface LogEntry {
  id: string
  command: string
  status: string
  createdAt: string
  outputText: string | null
}

interface AiLogData {
  logs?: LogEntry[]
}

export default function AiVisaCoachPage() {
  const session = getSession()
  const refId = session?.user.referenceId || ''
  const userPhone = session?.user.phone || ''
  const userEmail = session?.user.email || ''

  const [running, setRunning] = useState<CoachAction | null>(null)
  const [result, setResult] = useState<{ action: CoachAction; data: Record<string, unknown> } | null>(null)
  const [errorResult, setErrorResult] = useState<string | null>(null)
  const [otpDialogOpen, setOtpDialogOpen] = useState(false)
  const [pendingAction, setPendingAction] = useState<CoachAction | null>(null)
  const [otpSending, setOtpSending] = useState(false)
  const [otpVerifying, setOtpVerifying] = useState(false)
  const [otpSent, setOtpSent] = useState(false)
  const [otpCode, setOtpCode] = useState('')
  const [otpError, setOtpError] = useState<string | null>(null)

  // AI Coach log (recent actions)
  const { data: logData, loading: logLoading } = usePortalData<AiLogData>({
    endpoint: '/api/ai',
    params: { limit: 10 },
    enabled: !!refId,
  })
  const recentLogs = (logData?.logs || []).filter((l) => l.command?.startsWith('visa_coach_')).slice(0, 6)

  async function runAction(action: CoachAction, otpVerified: boolean = false) {
    if (!refId) {
      toast.error('Session not loaded. Please refresh.')
      return
    }
    setRunning(action)
    setErrorResult(null)
    setResult(null)
    try {
      const token = session?.token
      const res = await fetch('/api/visa/ai-coach', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${token}` },
        body: JSON.stringify({ studentId: refId, action, otpVerified }),
      })
      const json = await res.json()
      if (res.status === 403 && json.error === 'OTP_REQUIRED') {
        // Trigger OTP flow
        setPendingAction(action)
        setOtpDialogOpen(true)
        setOtpSent(false)
        setOtpCode('')
        setOtpError(null)
        setRunning(null)
        return
      }
      if (!res.ok) throw new Error(json.error || json.message || 'AI Coach request failed')
      setResult({ action, data: json })
      toast.success(`AI Coach: ${ACTIONS.find((a) => a.key === action)?.title} completed.`)
    } catch (e: unknown) {
      const msg = e instanceof Error ? e.message : 'AI Coach failed'
      setErrorResult(msg)
      toast.error(msg)
    } finally {
      setRunning(null)
    }
  }

  async function sendOtp() {
    setOtpSending(true)
    setOtpError(null)
    try {
      const token = session?.token
      const identifier = userPhone || userEmail
      if (!identifier) {
        throw new Error('No phone or email on session for OTP.')
      }
      const res = await fetch('/api/otp', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${token}` },
        body: JSON.stringify(
          userPhone
            ? { phone: userPhone, purpose: 'portal_access' }
            : { email: userEmail, purpose: 'portal_access' }
        ),
      })
      const json = await res.json()
      if (!res.ok) throw new Error(json.error || 'Failed to send OTP')
      setOtpSent(true)
      toast.success(`OTP sent to ${identifier}. Check your console for the dev code.`)
    } catch (e: unknown) {
      const msg = e instanceof Error ? e.message : 'Failed to send OTP'
      setOtpError(msg)
      toast.error(msg)
    } finally {
      setOtpSending(false)
    }
  }

  async function verifyOtpAndRun() {
    if (!pendingAction) return
    if (!otpCode || otpCode.length !== 6) {
      setOtpError('Please enter the 6-digit OTP code.')
      return
    }
    setOtpVerifying(true)
    setOtpError(null)
    try {
      const token = session?.token
      const res = await fetch('/api/otp/verify', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${token}` },
        body: JSON.stringify(
          userPhone
            ? { phone: userPhone, otp: otpCode, purpose: 'portal_access' }
            : { email: userEmail, otp: otpCode, purpose: 'portal_access' }
        ),
      })
      const json = await res.json()
      if (!res.ok || !json.success) {
        throw new Error(json.error || 'Invalid OTP code')
      }
      toast.success('OTP verified. Running action...')
      setOtpDialogOpen(false)
      setOtpCode('')
      setOtpSent(false)
      const action = pendingAction
      setPendingAction(null)
      await runAction(action, true)
    } catch (e: unknown) {
      const msg = e instanceof Error ? e.message : 'OTP verification failed'
      setOtpError(msg)
      toast.error(msg)
    } finally {
      setOtpVerifying(false)
    }
  }

  function renderResult() {
    if (errorResult) {
      return (
        <Alert className="border-red-200 bg-red-50">
          <AlertCircle className="h-4 w-4 text-red-600" />
          <AlertTitle className="text-red-800">AI Coach Error</AlertTitle>
          <AlertDescription className="text-red-800/90 text-sm">{errorResult}</AlertDescription>
        </Alert>
      )
    }
    if (!result) {
      return (
        <div className="text-center py-10">
          <Bot className="h-12 w-12 text-slate-300 mx-auto mb-3" />
          <p className="text-sm text-slate-500">Select an action above to get started.</p>
          <p className="text-xs text-slate-400 mt-1">Your AI Coach result will appear here.</p>
        </div>
      )
    }
    const { action, data } = result
    const def = ACTIONS.find((a) => a.key === action)
    const ResultIcon = def?.icon || Bot

    return (
      <div className="space-y-4">
        <div className="flex items-center gap-3">
          <div className={cn('h-10 w-10 rounded-lg flex items-center justify-center', def?.bg)}>
            <ResultIcon className={cn('h-5 w-5', def?.color)} />
          </div>
          <div>
            <p className="text-sm font-bold text-slate-900">{def?.title}</p>
            <p className="text-xs text-slate-500">Generated just now</p>
          </div>
        </div>
        <Separator />
        <RenderActionResult action={action} data={data} />
      </div>
    )
  }

  return (
    <div className="space-y-6">
      <PageHeader
        title="AI Visa Coach"
        description="AI-powered suggestions to strengthen your visa application. Suggestions only — not a guarantee."
      />

      {/* Compliance warning */}
      <Alert className="border-amber-200 bg-amber-50/60">
        <ShieldAlert className="h-4 w-4 text-amber-600" />
        <AlertTitle className="text-amber-800">AI Coach — Suggestions Only</AlertTitle>
        <AlertDescription className="text-amber-800/90 text-sm">
          The AI Visa Coach provides suggestions only. It cannot guarantee visa approval, submit applications, approve documents, or bypass staff approval.
        </AlertDescription>
      </Alert>

      {/* Stats */}
      <div className="grid grid-cols-2 lg:grid-cols-4 gap-4">
        <StatCard label="AI Actions" value={9} icon={Bot} accent="blue" hint="Available" />
        <StatCard label="OTP-Protected" value={3} icon={Lock} accent="orange" hint="Private data" />
        <StatCard label="Recent Runs" value={recentLogs.length} icon={Clock} accent="green" hint="Last 6" />
        <StatCard label="Suggestions" value={result ? 1 : 0} icon={Sparkles} accent="purple" hint="This session" />
      </div>

      <div className="grid lg:grid-cols-3 gap-6">
        {/* Action grid */}
        <div className="lg:col-span-2 space-y-4">
          <Card className="p-6 border-0 shadow-sm">
            <div className="mb-4">
              <h3 className="text-lg font-bold text-slate-900">AI Actions</h3>
              <p className="text-xs text-slate-500 mt-0.5">Click any action to generate AI-powered insights for your visa file.</p>
            </div>
            <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
              {ACTIONS.map((a) => {
                const Icon = a.icon
                const isRunning = running === a.key
                return (
                  <button
                    key={a.key}
                    onClick={() => runAction(a.key)}
                    disabled={!!running}
                    className={cn(
                      'group text-left p-4 rounded-lg border bg-white hover:shadow-md transition-all h-full',
                      'border-slate-200 hover:border-slate-300',
                      running && !isRunning && 'opacity-50 cursor-not-allowed',
                      isRunning && 'ring-2 ring-[#0057B8]/30',
                    )}
                  >
                    <div className="flex items-start justify-between gap-2 mb-2">
                      <div className={cn('h-10 w-10 rounded-lg flex items-center justify-center', a.bg)}>
                        {isRunning ? <RefreshCw className={cn('h-5 w-5 animate-spin', a.color)} /> : <Icon className={cn('h-5 w-5', a.color)} />}
                      </div>
                      {a.requiresOtp && (
                        <Badge variant="outline" className="text-[10px] bg-amber-50 text-amber-700 border-amber-200">
                          <Lock className="h-2.5 w-2.5 mr-0.5" /> OTP
                        </Badge>
                      )}
                    </div>
                    <p className="text-sm font-semibold text-slate-900">{a.title}</p>
                    <p className="text-xs text-slate-500 mt-1">{a.description}</p>
                  </button>
                )
              })}
            </div>
          </Card>

          {/* Result */}
          <Card className="p-6 border-0 shadow-sm">
            <h3 className="text-lg font-bold text-slate-900 mb-4">AI Coach Result</h3>
            {running ? (
              <div className="space-y-3">
                <Skeleton className="h-8 w-48" />
                <Skeleton className="h-4 w-full" />
                <Skeleton className="h-4 w-3/4" />
                <Skeleton className="h-32 w-full" />
              </div>
            ) : renderResult()}
          </Card>
        </div>

        {/* Sidebar: Recent runs */}
        <div className="space-y-4">
          <Card className="p-6 border-0 shadow-sm">
            <div className="flex items-center gap-2 mb-4">
              <Clock className="h-4 w-4 text-[#0057B8]" />
              <h3 className="text-sm font-bold text-slate-900">AI Coach Log</h3>
            </div>
            {logLoading ? (
              <div className="space-y-2">
                {[1, 2, 3, 4].map((i) => <Skeleton key={i} className="h-12 rounded-lg" />)}
              </div>
            ) : recentLogs.length === 0 ? (
              <div className="text-center py-6">
                <Bot className="h-8 w-8 text-slate-300 mx-auto mb-2" />
                <p className="text-xs text-slate-500">No AI actions run yet.</p>
                <p className="text-[11px] text-slate-400 mt-0.5">Your activity will appear here.</p>
              </div>
            ) : (
              <div className="space-y-2 max-h-[400px] overflow-y-auto alo-scrollbar pr-1">
                {recentLogs.map((log) => {
                  const actionName = log.command.replace('visa_coach_', '').replace(/_/g, ' ')
                  return (
                    <div key={log.id} className="p-3 rounded-lg border border-slate-100 bg-[#F8FAFC]">
                      <div className="flex items-center justify-between gap-2">
                        <p className="text-xs font-semibold text-slate-900 capitalize truncate">{actionName}</p>
                        <Badge variant="outline" className="text-[10px] bg-green-50 text-green-700 border-green-200 shrink-0">
                          <CheckCircle2 className="h-2.5 w-2.5 mr-0.5" /> Done
                        </Badge>
                      </div>
                      <p className="text-[11px] text-slate-400 mt-1">
                        {new Date(log.createdAt).toLocaleString('en-GB')}
                      </p>
                    </div>
                  )
                })}
              </div>
            )}
          </Card>

          <Card className="p-6 border-0 shadow-sm bg-[#F8FAFC]">
            <div className="flex items-center gap-2 mb-2">
              <Sparkles className="h-4 w-4 text-[#E8501A]" />
              <p className="text-sm font-bold text-slate-900">How It Works</p>
            </div>
            <ol className="text-xs text-slate-600 space-y-2 list-decimal list-inside">
              <li>Click an action card to run it.</li>
              <li>OTP-protected actions prompt for a 6-digit code.</li>
              <li>OTP is sent to your registered phone or email.</li>
              <li>Verify and the action runs automatically.</li>
              <li>Results appear in the panel on the left.</li>
            </ol>
          </Card>
        </div>
      </div>

      {/* OTP Dialog */}
      <Dialog open={otpDialogOpen} onOpenChange={setOtpDialogOpen}>
        <DialogContent className="sm:max-w-md">
          <DialogHeader>
            <DialogTitle className="flex items-center gap-2">
              <Lock className="h-5 w-5 text-[#E8501A]" /> OTP Verification Required
            </DialogTitle>
            <DialogDescription>
[REDACTED: potential embedded credential in legacy reference]
            </DialogDescription>
          </DialogHeader>

          <div className="space-y-4 py-2">
            {!otpSent ? (
              <div className="space-y-3">
                <div className="flex items-center gap-3 p-3 rounded-lg bg-[#F8FAFC] border border-slate-200">
                  <div className="h-9 w-9 rounded-lg bg-[#EFF6FF] flex items-center justify-center shrink-0">
                    <Phone className="h-4 w-4 text-[#0057B8]" />
                  </div>
                  <div className="flex-1 min-w-0">
                    <p className="text-xs text-slate-500">OTP will be sent to</p>
                    <p className="text-sm font-semibold text-slate-900 truncate">{userPhone || userEmail || '—'}</p>
                  </div>
                </div>
                <Button onClick={sendOtp} disabled={otpSending} className="w-full h-12 bg-[#0057B8] hover:bg-[#004a99]">
                  {otpSending ? <><RefreshCw className="h-4 w-4 mr-2 animate-spin" /> Sending...</> : <><KeyRound className="h-4 w-4 mr-2" /> Send OTP</>}
                </Button>
              </div>
            ) : (
              <div className="space-y-3">
                <div className="space-y-2">
                  <Label htmlFor="otp-code" className="text-sm font-medium">Enter 6-digit OTP</Label>
                  <Input
                    id="otp-code"
                    type="text"
                    inputMode="numeric"
                    pattern="[0-9]*"
                    maxLength={6}
                    placeholder="••••••"
                    value={otpCode}
                    onChange={(e) => setOtpCode(e.target.value.replace(/\D/g, ''))}
                    className="h-12 text-center text-lg tracking-[0.5em] font-mono"
                  />
                </div>
                {otpError && (
                  <p className="text-xs text-red-600 flex items-center gap-1">
                    <AlertCircle className="h-3 w-3" /> {otpError}
                  </p>
                )}
                <Button onClick={verifyOtpAndRun} disabled={otpVerifying || otpCode.length !== 6} className="w-full h-12 bg-[#E8501A] hover:bg-[#c44314]">
                  {otpVerifying ? <><RefreshCw className="h-4 w-4 mr-2 animate-spin" /> Verifying...</> : <><CheckCircle2 className="h-4 w-4 mr-2" /> Verify & Run</>}
                </Button>
                <button
                  onClick={sendOtp}
                  disabled={otpSending}
                  className="text-xs text-[#0057B8] hover:underline w-full text-center"
                >
                  Resend OTP
                </button>
              </div>
            )}
          </div>

          <DialogFooter className="text-xs text-slate-400">
            <p>OTP expires in 10 minutes. Max 5 verification attempts.</p>
          </DialogFooter>
        </DialogContent>
      </Dialog>
    </div>
  )
}

function RenderActionResult({ action, data }: { action: CoachAction; data: Record<string, unknown> }) {
  switch (action) {
    case 'summarize_file': {
      const summary = (data.summary || {}) as Record<string, unknown>
      const rows = Object.entries(summary).map(([k, v]) => ({ k, v }))
      return (
        <div>
          <p className="text-sm font-semibold text-slate-900 mb-3">File Summary</p>
          <div className="grid grid-cols-2 gap-2">
            {rows.map((r) => (
              <div key={r.k} className="p-3 rounded-lg bg-[#F8FAFC] border border-slate-100">
                <p className="text-[11px] text-slate-500 uppercase tracking-wide">{r.k.replace(/([A-Z])/g, ' $1').trim()}</p>
                <p className="text-sm font-bold text-slate-900 mt-0.5">{String(r.v)}</p>
              </div>
            ))}
          </div>
        </div>
      )
    }
    case 'detect_missing_documents': {
      const missing = (data.missing || []) as string[]
      const expired = (data.expired || []) as Array<Record<string, unknown>>
      return (
        <div className="space-y-4">
          <div>
            <p className="text-sm font-semibold text-slate-900 mb-2">Missing Items ({missing.length})</p>
            {missing.length === 0 ? (
              <p className="text-sm text-green-600">No missing items detected.</p>
            ) : (
              <ul className="space-y-1.5">
                {missing.map((m, i) => (
                  <li key={i} className="text-sm text-slate-700 flex items-start gap-2">
                    <AlertTriangle className="h-4 w-4 text-amber-500 mt-0.5 shrink-0" /> {m}
                  </li>
                ))}
              </ul>
            )}
          </div>
          <Separator />
          <div>
            <p className="text-sm font-semibold text-slate-900 mb-2">Expired Documents ({expired.length})</p>
            {expired.length === 0 ? (
              <p className="text-sm text-green-600">No expired documents.</p>
            ) : (
              <ul className="space-y-1.5">
                {expired.map((d, i) => (
                  <li key={i} className="text-sm text-slate-700 flex items-start gap-2">
                    <AlertCircle className="h-4 w-4 text-red-500 mt-0.5 shrink-0" />
                    <span>{String(d.documentType || 'document')}: <span className="text-slate-500">{String(d.fileName || '')}</span></span>
                  </li>
                ))}
              </ul>
            )}
          </div>
        </div>
      )
    }
    case 'suggest_next_action': {
      const actions = (data.nextActions || []) as string[]
      return (
        <div>
          <p className="text-sm font-semibold text-slate-900 mb-3">Suggested Next Actions</p>
          <ol className="space-y-2">
            {actions.map((a, i) => (
              <li key={i} className="flex items-start gap-3 p-3 rounded-lg bg-[#F8FAFC] border border-slate-100">
                <span className="h-6 w-6 rounded-full bg-[#0057B8] text-white text-xs font-bold flex items-center justify-center shrink-0">
                  {i + 1}
                </span>
                <span className="text-sm text-slate-700">{a}</span>
              </li>
            ))}
          </ol>
        </div>
      )
    }
    case 'create_interview_questions': {
      const questions = (data.questions || []) as Array<Record<string, string>>
      return (
        <div>
          <p className="text-sm font-semibold text-slate-900 mb-3">AI-Generated Interview Questions ({questions.length})</p>
          <div className="space-y-2">
            {questions.map((q, i) => (
              <div key={i} className="p-3 rounded-lg bg-[#F8FAFC] border border-slate-100">
                <Badge variant="outline" className="text-[10px] bg-[#EFF6FF] text-[#0057B8] border-[#BED9F5] mb-1">
                  {q.category?.replace(/_/g, ' ') || 'general'}
                </Badge>
                <p className="text-sm text-slate-800">{q.question}</p>
              </div>
            ))}
          </div>
        </div>
      )
    }
    case 'draft_sop': {
      const sop = String(data.draftSop || '')
      const note = String(data.note || '')
      return (
        <div className="space-y-3">
          <p className="text-sm font-semibold text-slate-900">AI-Drafted SOP</p>
          <div className="p-4 rounded-lg bg-[#F8FAFC] border border-slate-200">
            <pre className="text-sm text-slate-700 whitespace-pre-wrap font-sans">{sop}</pre>
          </div>
          {note && (
            <Alert className="border-amber-200 bg-amber-50">
              <AlertCircle className="h-4 w-4 text-amber-600" />
              <AlertDescription className="text-amber-800/90 text-sm">{note}</AlertDescription>
            </Alert>
          )}
          <Button asChild size="sm" variant="outline" className="h-10">
            <Link href="/student/ai-sop">Open AI SOP Builder <ArrowRight className="h-3 w-3 ml-1" /></Link>
          </Button>
        </div>
      )
    }
    case 'highlight_financial_risk': {
      const risk = (data.financialRisk || {}) as Record<string, unknown>
      const missingItems = (risk.missingItems || []) as string[]
      const level = String(risk.riskFlag || 'unknown')
      const levelMeta: Record<string, { color: string; bg: string; label: string }> = {
        low: { color: 'text-green-700', bg: 'bg-green-50 border-green-200', label: 'Low Risk' },
        medium: { color: 'text-amber-700', bg: 'bg-amber-50 border-amber-200', label: 'Medium Risk' },
        high: { color: 'text-red-700', bg: 'bg-red-50 border-red-200', label: 'High Risk' },
        unknown: { color: 'text-slate-700', bg: 'bg-slate-50 border-slate-200', label: 'Not Assessed' },
      }
      const meta = levelMeta[level] || levelMeta.unknown
      return (
        <div className="space-y-4">
          <div className={cn('p-3 rounded-lg border', meta.bg)}>
            <p className={cn('text-sm font-bold', meta.color)}>{meta.label}</p>
          </div>
          <div>
            <p className="text-sm font-semibold text-slate-900 mb-2">Missing / Unapproved Items ({missingItems.length})</p>
            {missingItems.length === 0 ? (
              <p className="text-sm text-green-600">All financial documents approved.</p>
            ) : (
              <div className="flex flex-wrap gap-2">
                {missingItems.map((m, i) => (
                  <Badge key={i} variant="outline" className="bg-red-50 text-red-700 border-red-200">{m}</Badge>
                ))}
              </div>
            )}
          </div>
          <Alert className="border-[#BED9F5] bg-[#EFF6FF]">
            <Lightbulb className="h-4 w-4 text-[#0057B8]" />
            <AlertDescription className="text-[#0057B8]/90 text-sm">{String(risk.recommendation || '')}</AlertDescription>
          </Alert>
        </div>
      )
    }
    case 'highlight_refusal_risk': {
      const risk = (data.refusalRisk || {}) as Record<string, unknown>
      const factors = (risk.factors || []) as string[]
      const level = String(risk.level || 'minimal')
      const levelMeta: Record<string, { color: string; bg: string }> = {
        minimal: { color: 'text-green-700', bg: 'bg-green-50 border-green-200' },
        low: { color: 'text-green-700', bg: 'bg-green-50 border-green-200' },
        medium: { color: 'text-amber-700', bg: 'bg-amber-50 border-amber-200' },
        high: { color: 'text-red-700', bg: 'bg-red-50 border-red-200' },
      }
      const meta = levelMeta[level] || levelMeta.minimal
      return (
        <div className="space-y-4">
          <div className={cn('p-3 rounded-lg border flex items-center justify-between', meta.bg)}>
            <p className={cn('text-sm font-bold capitalize', meta.color)}>Refusal Risk: {level}</p>
            <span className="text-xs">{factors.length} factor(s)</span>
          </div>
          {factors.length === 0 ? (
            <p className="text-sm text-green-600">No major refusal risk factors detected.</p>
          ) : (
            <ul className="space-y-2">
              {factors.map((f, i) => (
                <li key={i} className="flex items-start gap-2 text-sm text-slate-700">
                  <AlertTriangle className="h-4 w-4 text-amber-500 mt-0.5 shrink-0" /> {f}
                </li>
              ))}
            </ul>
          )}
          <Alert className="border-[#BED9F5] bg-[#EFF6FF]">
            <Lightbulb className="h-4 w-4 text-[#0057B8]" />
            <AlertDescription className="text-[#0057B8]/90 text-sm">{String(risk.recommendation || '')}</AlertDescription>
          </Alert>
        </div>
      )
    }
    case 'generate_counselor_summary': {
      const text = String(data.counselorSummary || '')
      return (
        <div>
          <p className="text-sm font-semibold text-slate-900 mb-3">Counselor Summary</p>
          <div className="p-4 rounded-lg bg-[#F8FAFC] border border-slate-200">
            <p className="text-sm text-slate-800">{text}</p>
          </div>
        </div>
      )
    }
    case 'generate_visa_officer_checklist': {
      const items = (data.visaOfficerChecklist || []) as Array<Record<string, unknown>>
      return (
        <div>
          <p className="text-sm font-semibold text-slate-900 mb-3">Visa Officer Checklist ({items.length})</p>
          <div className="overflow-x-auto">
            <table className="w-full text-sm">
              <thead>
                <tr className="text-left text-xs text-slate-500 uppercase border-b border-slate-200">
                  <th className="py-2 pr-3">Item</th>
                  <th className="py-2 px-3">Weight</th>
                  <th className="py-2 px-3">Status</th>
                  <th className="py-2 pl-3">Action</th>
                </tr>
              </thead>
              <tbody>
                {items.map((it, i) => (
                  <tr key={i} className="border-b border-slate-100">
                    <td className="py-2.5 pr-3 font-medium text-slate-800">{String(it.item || '')}</td>
                    <td className="py-2.5 px-3 text-slate-600">{String(it.weight || '')}%</td>
                    <td className="py-2.5 px-3 capitalize text-slate-600">{String(it.status || '').replace(/_/g, ' ')}</td>
                    <td className="py-2.5 pl-3">
                      <Badge variant="outline" className="text-[10px] bg-slate-50 text-slate-600 border-slate-200">
                        {String(it.action || '')}
                      </Badge>
                    </td>
                  </tr>
                ))}
              </tbody>
            </table>
          </div>
        </div>
      )
    }
    default:
      return <pre className="text-xs text-slate-600 whitespace-pre-wrap">{JSON.stringify(data, null, 2)}</pre>
  }
}
