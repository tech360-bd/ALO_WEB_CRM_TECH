// ----------------------------------------------------------
// Helper: Timing-safe OTP hash verification
// ----------------------------------------------------------
function verifyOtpHash(otp: string, hash: string): boolean {
  const computed = hashOtp(otp);
  try {
    return crypto.timingSafeEqual(Buffer.from(computed), Buffer.from(hash));
  } catch {
    return false;
  }
}
