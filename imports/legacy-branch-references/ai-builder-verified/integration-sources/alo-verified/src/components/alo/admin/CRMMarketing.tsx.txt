// ─── Component ───
export default function CRMMarketing() {
  const [activeTab, setActiveTab] = useState('dashboard')
  const [ideas, setIdeas] = useState(INIT_IDEAS)
  const [campaigns, setCampaigns] = useState<any[]>([])
  const [content, setContent] = useState<any[]>([])
  const [socialSizes, setSocialSizes] = useState<any[]>([])
  const [scheduled, setScheduled] = useState(INIT_SCHEDULED)
  const [prompts, setPrompts] = useState(INIT_PROMPTS)
  const [scripts, setScripts] = useState(INIT_SCRIPTS)
  const [approvals, setApprovals] = useState(INIT_APPROVALS)
  const [platformFilter, setPlatformFilter] = useState('all')
  const [genIdeas, setGenIdeas] = useState<string[]>([])
  const [showGenIdeas, setShowGenIdeas] = useState(false)
  const [genContent, setGenContent] = useState('')
  const [genForm, setGenForm] = useState({ type: 'social_post', platform: 'instagram', topic: '' })
  const [expandedIdea, setExpandedIdea] = useState<string | null>(null)
  const [selectedApproval, setSelectedApproval] = useState<string | null>(null)
  const [loading, setLoading] = useState(true)

  useEffect(() => {
    const fetchMarketingData = async () => {
      try {
        setLoading(true)
        const res = await fetch('/api/marketing')
        if (res.ok) {
          const data = await res.json()
          setCampaigns(data.campaigns || [])
          setContent(data.content || [])
          setSocialSizes(data.socialSizes || [])
        }
      } catch (err) {
        console.error('Failed to fetch marketing data:', err)
      } finally {
        setLoading(false)
      }
    }
    fetchMarketingData()
  }, [])

  // Dialogs
  const [ideaDialog, setIdeaDialog] = useState(false)
  const [campaignDialog, setCampaignDialog] = useState(false)
  const [contentDialog, setContentDialog] = useState(false)
  const [promptDialog, setPromptDialog] = useState(false)
  const [scriptDialog, setScriptDialog] = useState(false)
  const [scheduleDialog, setScheduleDialog] = useState(false)

  // Forms
  const [ideaForm, setIdeaForm] = useState({ title: '', description: '', category: 'social', priority: 'medium', tags: '' })
  const [campaignForm, setCampaignForm] = useState({ name: '', type: 'social_media', startDate: '', endDate: '', budget: '' })
  const [contentForm, setContentForm] = useState({ title: '', type: 'social_post', platform: 'instagram', assignedTo: '', campaignId: '' })
  const [promptForm, setPromptForm] = useState({ prompt: '', negativePrompt: '', style: 'realistic', platform: 'instagram', size: '1080×1080' })
  const [scriptForm, setScriptForm] = useState({ title: '', script: '', duration: '60s', platform: 'tiktok', hookLine: '', cta: '' })
  const [scheduleForm, setScheduleForm] = useState({ contentId: '', date: '', time: '10:00' })

  const addIdea = () => {
    if (!ideaForm.title) return toast.error('Title required')
    setIdeas(prev => [...prev, { id: `i${Date.now()}`, ...ideaForm, source: 'manual', status: 'idea', assignedTo: '', tags: ideaForm.tags.split(',').map(t => t.trim()).filter(Boolean) }])
    setIdeaForm({ title: '', description: '', category: 'social', priority: 'medium', tags: '' })
    setIdeaDialog(false)
    toast.success('Idea added')
  }

  const addCampaign = () => {
    if (!campaignForm.name) return toast.error('Name required')
    toast.success('Campaign created')
    setCampaignForm({ name: '', type: 'social_media', startDate: '', endDate: '', budget: '' })
    setCampaignDialog(false)
  }

  const addContent = () => {
    if (!contentForm.title) return toast.error('Title required')
    setContent(prev => [...prev, { id: `co${Date.now()}`, ...contentForm, status: 'draft', campaignId: contentForm.campaignId || '' }])
    setContentForm({ title: '', type: 'social_post', platform: 'instagram', assignedTo: '', campaignId: '' })
    setContentDialog(false)
    toast.success('Content created')
  }

  const addPrompt = () => {
    if (!promptForm.prompt) return toast.error('Prompt required')
    setPrompts(prev => [...prev, { id: `p${Date.now()}`, ...promptForm, status: 'draft' }])
    setPromptForm({ prompt: '', negativePrompt: '', style: 'realistic', platform: 'instagram', size: '1080×1080' })
    setPromptDialog(false)
    toast.success('Image prompt created')
  }

  const addScript = () => {
    if (!scriptForm.title) return toast.error('Title required')
    setScripts(prev => [...prev, { id: `v${Date.now()}`, ...scriptForm, status: 'draft' }])
    setScriptForm({ title: '', script: '', duration: '60s', platform: 'tiktok', hookLine: '', cta: '' })
    setScriptDialog(false)
    toast.success('Video script created')
  }

  const addSchedule = () => {
    if (!scheduleForm.contentId || !scheduleForm.date) return toast.error('Select content and date')
    const c = content.find(x => x.id === scheduleForm.contentId)
    if (!c) return
    setScheduled(prev => [...prev, { id: `s${Date.now()}`, contentId: c.id, platform: c.platform, title: c.title, date: scheduleForm.date, time: scheduleForm.time, status: 'scheduled' }])
    setScheduleDialog(false)
    toast.success('Post scheduled')
  }

  const generateIdeas = () => {
    setGenIdeas([
      '🎓 "5 Mistakes Students Make When Applying Abroad" - Blog series targeting common application errors',
      '📱 "Campus Tour Tuesday" - Weekly social series showing different university campuses',
      '💬 "WhatsApp Quiz: Which Country Suits You?" - Interactive lead generation quiz',
      '🎬 "60-Second Study Hack" - TikTok/Reels series with quick study tips for IELTS/TOEFL',
      '📧 "Scholarship Alert Newsletter" - Monthly email highlighting new scholarship deadlines'
    ])
    setShowGenIdeas(true)
  }

  const generateContent = () => {
    if (!genForm.topic) return toast.error('Enter a topic')
    const templates: Record<string, string> = {
      social_post: `✨ ${genForm.topic} ✨\n\nDreaming of studying abroad? Here is your chance! 🌍\n\n🔹 World-class education\n🔹 Global networking\n🔹 Career opportunities\n\n📩 DM us "INFO" to get started!\n\n#StudyAbroad #${genForm.platform === 'tiktok' ? 'TikTokStudy' : 'Education'} #InternationalStudents`,
      blog_post: `# ${genForm.topic}\n\n## Introduction\nThe journey of ${genForm.topic.toLowerCase()} is one that many Bangladeshi students dream about. In this comprehensive guide, we will walk you through everything you need to know.\n\n## Why Choose This Path?\n- Quality education standards\n- Affordable living costs\n- Post-study work opportunities\n\n## How to Get Started\n1. Research your options\n2. Prepare your documents\n3. Apply through ALO Education\n\n## Conclusion\nReady to take the first step? Contact us today!`,
      email_template: `Subject: 🎓 ${genForm.topic} - Your Journey Starts Here!\n\nDear {{name}},\n\nAre you ready to explore ${genForm.topic.toLowerCase()}? We have exciting opportunities waiting for you!\n\n✅ Expert guidance every step of the way\n✅ Scholarship assistance\n✅ Visa support\n\n[Apply Now Button]\n\nBest regards,\nALO Education Team`,
      whatsapp_template: `🎓 *${genForm.topic}*\n\nHi {{name}}! Great news about ${genForm.topic.toLowerCase()}!\n\n✅ Expert counseling\n✅ Scholarship help\n✅ Visa guidance\n\nReply YES to learn more! 📩`,
      image_script: `🖼️ Image Concept: ${genForm.topic}\n\nVisual: Modern university campus with diverse students\nOverlay text: "${genForm.topic}"\nStyle: Clean, professional, inspiring\n\nSuggested size: ${SIZES[genForm.platform] || '1080×1080'}px`,
[REDACTED: potential embedded credential in legacy reference]
    }
    setGenContent(templates[genForm.type] || templates.social_post)
    toast.success('Content generated!')
  }

  const approveIdea = (id: string) => { setIdeas(prev => prev.map(i => i.id === id ? { ...i, status: 'approved' } : i)); toast.success('Idea approved') }
  const rejectIdea = (id: string) => { setIdeas(prev => prev.map(i => i.id === id ? { ...i, status: 'rejected' } : i)); toast.success('Idea rejected') }

  const handleApproval = (id: string, action: 'approved' | 'rejected' | 'changes') => {
    setApprovals(prev => prev.filter(a => a.id !== id))
    if (action === 'approved') toast.success('Content approved & added to schedule')
    else if (action === 'rejected') toast.success('Content rejected')
    else toast.success('Changes requested')
    setSelectedApproval(null)
  }

  const pipelineStages = [
    { label: 'Ideas', count: ideas.filter(i => i.status === 'idea').length, color: 'bg-blue-500' },
    { label: 'In Progress', count: content.filter(c => c.status === 'in_progress').length, color: 'bg-amber-500' },
    { label: 'Review', count: content.filter(c => c.status === 'review').length, color: 'bg-purple-500' },
    { label: 'Approved', count: content.filter(c => c.status === 'approved').length, color: 'bg-emerald-500' },
    { label: 'Scheduled', count: scheduled.filter(s => s.status === 'scheduled').length, color: 'bg-sky-500' },
    { label: 'Published', count: scheduled.filter(s => s.status === 'published').length, color: 'bg-green-500' },
  ]

  const filteredScheduled = platformFilter === 'all' ? scheduled : scheduled.filter(s => s.platform === platformFilter)

  return (
    <div className="space-y-4">
      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-3">
        <div>
          <h1 className="text-2xl font-bold text-[#1A2332]">Marketing Studio</h1>
          <p className="text-sm text-gray-500">Idea → Campaign → Content → Approval → Schedule → Publish</p>
        </div>
        <div className="flex gap-2 flex-wrap">
          <Button variant="outline" size="sm" onClick={() => { setIdeaDialog(true) }} className="border-[#F37021] text-[#F37021]"><Lightbulb className="h-4 w-4 mr-1" /> New Idea</Button>
          <Button variant="outline" size="sm" onClick={() => { setCampaignDialog(true) }} className="border-[#0066CC] text-[#0066CC]"><Megaphone className="h-4 w-4 mr-1" /> New Campaign</Button>
          <Button size="sm" className="bg-[#F37021] hover:bg-[#e06010] text-white"><Sparkles className="h-4 w-4 mr-1" /> AI Generate</Button>
        </div>
      </div>

      <Tabs value={activeTab} onValueChange={setActiveTab}>
        <ScrollArea className="w-full">
          <TabsList className="bg-gray-100 mb-4 flex w-max min-w-full">
            <TabsTrigger value="dashboard" className="text-xs sm:text-sm"><LayoutGrid className="h-3 w-3 mr-1" /> Dashboard</TabsTrigger>
            <TabsTrigger value="ideas" className="text-xs sm:text-sm"><Lightbulb className="h-3 w-3 mr-1" /> Ideas</TabsTrigger>

                          <Badge className={`text-[10px] ${a.riskLevel === 'high' ? 'bg-red-100 text-red-700' : a.riskLevel === 'medium' ? 'bg-amber-100 text-amber-700' : 'bg-green-100 text-green-700'}`}>{a.riskLevel} risk</Badge>
                        </div>
                      </div>
                      <Separator />
                      <div className="bg-gray-50 rounded-lg p-3 max-h-48 overflow-y-auto">
                        <pre className="text-xs whitespace-pre-wrap font-sans">{a.content}</pre>
                      </div>
                      <div className="text-[10px] text-gray-400">Requested by {a.requestedBy} on {a.date}</div>
                      <div className="flex gap-2">
                        <Button size="sm" className="flex-1 bg-green-600 hover:bg-green-700 text-white h-8 text-xs" onClick={() => handleApproval(a.id, 'approved')}><CheckCircle className="h-3 w-3 mr-1" /> Approve</Button>
                        <Button size="sm" variant="outline" className="flex-1 h-8 text-xs text-red-600 border-red-300" onClick={() => handleApproval(a.id, 'rejected')}><ThumbsDown className="h-3 w-3 mr-1" /> Reject</Button>
                      </div>
                      <Button size="sm" variant="outline" className="w-full h-8 text-xs text-amber-600 border-amber-300" onClick={() => handleApproval(a.id, 'changes')}><MessageSquare className="h-3 w-3 mr-1" /> Request Changes</Button>
                    </div>
                  )
                })() : <p className="text-sm text-gray-400 text-center py-8">Select an item to preview</p>}
              </CardContent>
            </Card>
          </div>
        </TabsContent>
      </Tabs>

      {/* ─── Dialogs ─── */}
      <Dialog open={ideaDialog} onOpenChange={setIdeaDialog}>
        <DialogContent className="max-w-md">
          <DialogHeader><DialogTitle>Add Idea</DialogTitle></DialogHeader>
          <div className="space-y-3">
            <div><Label className="text-xs">Title *</Label><Input className="h-8 text-sm" value={ideaForm.title} onChange={e => setIdeaForm({ ...ideaForm, title: e.target.value })} /></div>
            <div><Label className="text-xs">Description</Label><Textarea className="text-sm" rows={2} value={ideaForm.description} onChange={e => setIdeaForm({ ...ideaForm, description: e.target.value })} /></div>
            <div className="grid grid-cols-2 gap-3">
              <div><Label className="text-xs">Category</Label>
                <Select value={ideaForm.category} onValueChange={v => setIdeaForm({ ...ideaForm, category: v })}>
                  <SelectTrigger className="h-8 text-sm"><SelectValue /></SelectTrigger>
                  <SelectContent>{['seo', 'social', 'email', 'whatsapp', 'blog', 'video'].map(c => <SelectItem key={c} value={c} className="capitalize">{c}</SelectItem>)}</SelectContent>
                </Select>
              </div>
              <div><Label className="text-xs">Priority</Label>
                <Select value={ideaForm.priority} onValueChange={v => setIdeaForm({ ...ideaForm, priority: v })}>
                  <SelectTrigger className="h-8 text-sm"><SelectValue /></SelectTrigger>
                  <SelectContent>{['high', 'medium', 'low'].map(p => <SelectItem key={p} value={p} className="capitalize">{p}</SelectItem>)}</SelectContent>
                </Select>
              </div>
            </div>
            <div><Label className="text-xs">Tags (comma separated)</Label><Input className="h-8 text-sm" value={ideaForm.tags} onChange={e => setIdeaForm({ ...ideaForm, tags: e.target.value })} placeholder="UK, scholarship" /></div>
            <Button className="w-full bg-[#F37021] hover:bg-[#e06010] text-white" onClick={addIdea}>Add Idea</Button>
          </div>
        </DialogContent>
      </Dialog>

      <Dialog open={campaignDialog} onOpenChange={setCampaignDialog}>
        <DialogContent className="max-w-md">
          <DialogHeader><DialogTitle>Create Campaign</DialogTitle></DialogHeader>
          <div className="space-y-3">
            <div><Label className="text-xs">Name *</Label><Input className="h-8 text-sm" value={campaignForm.name} onChange={e => setCampaignForm({ ...campaignForm, name: e.target.value })} /></div>
            <div><Label className="text-xs">Type</Label>
              <Select value={campaignForm.type} onValueChange={v => setCampaignForm({ ...campaignForm, type: v })}>

                <SelectContent>{['social_media', 'email', 'whatsapp', 'blog_series', 'multi_channel'].map(t => <SelectItem key={t} value={t} className="capitalize">{t.replace('_', ' ')}</SelectItem>)}</SelectContent>
              </Select>
            </div>
            <div className="grid grid-cols-2 gap-3">
              <div><Label className="text-xs">Start Date</Label><Input type="date" className="h-8 text-sm" value={campaignForm.startDate} onChange={e => setCampaignForm({ ...campaignForm, startDate: e.target.value })} /></div>
              <div><Label className="text-xs">End Date</Label><Input type="date" className="h-8 text-sm" value={campaignForm.endDate} onChange={e => setCampaignForm({ ...campaignForm, endDate: e.target.value })} /></div>
            </div>
            <div><Label className="text-xs">Budget</Label><Input type="number" className="h-8 text-sm" value={campaignForm.budget} onChange={e => setCampaignForm({ ...campaignForm, budget: e.target.value })} placeholder="$0" /></div>
            <Button className="w-full bg-[#0066CC] hover:bg-[#0055aa] text-white" onClick={addCampaign}>Create Campaign</Button>
          </div>
        </DialogContent>
      </Dialog>

      <Dialog open={contentDialog} onOpenChange={setContentDialog}>
        <DialogContent className="max-w-md">
          <DialogHeader><DialogTitle>Create Content</DialogTitle></DialogHeader>
          <div className="space-y-3">
            <div><Label className="text-xs">Title *</Label><Input className="h-8 text-sm" value={contentForm.title} onChange={e => setContentForm({ ...contentForm, title: e.target.value })} /></div>
            <div className="grid grid-cols-2 gap-3">
              <div><Label className="text-xs">Type</Label>
                <Select value={contentForm.type} onValueChange={v => setContentForm({ ...contentForm, type: v })}>
                  <SelectTrigger className="h-8 text-sm"><SelectValue /></SelectTrigger>
                  <SelectContent>{['social_post', 'blog_post', 'email_template', 'whatsapp_template', 'image', 'video_script'].map(t => <SelectItem key={t} value={t} className="capitalize">{t.replace('_', ' ')}</SelectItem>)}</SelectContent>
                </Select>
              </div>
              <div><Label className="text-xs">Platform</Label>
                <Select value={contentForm.platform} onValueChange={v => setContentForm({ ...contentForm, platform: v })}>
                  <SelectTrigger className="h-8 text-sm"><SelectValue /></SelectTrigger>
                  <SelectContent>{Object.entries(PL).map(([k, v]) => <SelectItem key={k} value={k}>{v}</SelectItem>)}</SelectContent>
                </Select>
              </div>
            </div>
            <div className="grid grid-cols-2 gap-3">
              <div><Label className="text-xs">Assigned To</Label><Input className="h-8 text-sm" value={contentForm.assignedTo} onChange={e => setContentForm({ ...contentForm, assignedTo: e.target.value })} /></div>
              <div><Label className="text-xs">Campaign</Label>
                <Select value={contentForm.campaignId} onValueChange={v => setContentForm({ ...contentForm, campaignId: v })}>
                  <SelectTrigger className="h-8 text-sm"><SelectValue placeholder="None" /></SelectTrigger>
                  <SelectContent><SelectItem value="none">None</SelectItem>{campaigns.map(c => <SelectItem key={c.id} value={c.id}>{c.name}</SelectItem>)}</SelectContent>

              </div>
            </div>

      </Dialog>

          <div className="space-y-3">

            </div>
            <Button className="w-full bg-[#F37021] hover:bg-[#e06010] text-white" onClick={addPrompt}>Create Prompt</Button>
          </div>

      </Dialog>

      <Dialog open={scriptDialog} onOpenChange={setScriptDialog}>
