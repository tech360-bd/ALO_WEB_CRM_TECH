/**
 * ALO Education - Centralized Public Data
 *
 * Single source of truth for university/course/destination data shown on
 * the public website.  This mirrors the seed data so the homepage,
 * course finder, and university listing pages always stay in sync.
 *
 * Real ALO partner universities are listed first (Greenwich, Coventry,
 * East London, De Montfort, Roehampton, BPP, Ulster, Portsmouth) —
 * matching the AI Studio reference build — followed by top global
 * universities for breadth.
 */

export interface AloUniversity {
  name: string
  slug: string
  country: string
  countryCode: string
  city: string
  ranking: number
  type: 'public' | 'private'
  logo: string
  courses: number
  tuitionFrom: number
  applicationFee: number
  deposit: number
  englishRequirement: string
  popularFor: string[]
  description: string
  established: number
  students: string
  featured: boolean
  partner: boolean
}

export interface AloCourse {
  name: string
  level: 'undergraduate' | 'postgraduate' | 'phd' | 'diploma'
  field: string
  duration: string
  tuitionFee: number
  popularCountries: string[]
  icon: string
  description: string
}

export interface AloDestination {
  country: string
  code: string
  flag: string
  students: string
  universities: string
  desc: string
  gradient: string
  popularFor: string[]
  intake: string
  workWhileStudy: string
  postStudyVisa: string
}
